```cpp
#include <vector>
using namespace std;

class Solution {
    vector<int> parent;

    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    bool unite(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;
        parent[rb] = ra;
        return true;
    }

public:
    vector<int> findRedundantDirectedConnection(vector<vector<int>>& edges) {
        int n = edges.size();
        vector<int> nodeParent(n + 1, 0);
        vector<int> cand1, cand2;

        for (const auto& e : edges) {
            int u = e[0], v = e[1];
            if (nodeParent[v] == 0) {
                nodeParent[v] = u;
            } else {
                cand1 = {nodeParent[v], v};
                cand2 = {u, v};
            }
        }

        parent.resize(n + 1);
        for (int i = 1; i <= n; ++i) parent[i] = i;

        for (const auto& e : edges) {
            if (!cand2.empty() && e == cand2) continue;
            if (!unite(e[0], e[1])) {
                return cand1.empty() ? e : cand1;
            }
        }

        return cand2;
    }
};
```
